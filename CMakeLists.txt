cmake_minimum_required(VERSION 3.10)
project(agri-robot-training VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_library(tractor_state_lib src/tractor_state.cpp)

add_executable(tcp_server src/tcp_server.cpp)
target_link_libraries(tcp_server tractor_state_lib)
add_executable(tcp_client src/tcp_client.cpp)
target_link_libraries(tcp_client tractor_state_lib)
